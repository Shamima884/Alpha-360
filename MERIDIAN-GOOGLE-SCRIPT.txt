/* ═══════════════════════ SPREADSHEET ACCESS ═══════════════════════ */

function ss(){
  var id = CONFIG.SPREADSHEET_ID ||
           PropertiesService.getScriptProperties().getProperty('MERIDIAN_SHEET_ID');
  if(id) return SpreadsheetApp.openById(id);
  var s = SpreadsheetApp.create('Meridian DB (auto-created)');
  PropertiesService.getScriptProperties().setProperty('MERIDIAN_SHEET_ID', s.getId());
  return s;
}

function sheetFor_(key){
  var book = ss();
  var name = CONFIG.SHEETS[key];
  var sh = book.getSheetByName(name);
  if(!sh) sh = book.insertSheet(name);
  return sh;
}

/* ═══════════════════════ READ / WRITE ═══════════════════════ */

function readCollection_(key){
  var sh = sheetFor_(key);
  var values = sh.getDataRange().getValues();
  if(values.length < 2) return [];
  var headers = values[0], rows = [];
  for(var i = 1; i < values.length; i++){
    var o = {}, empty = true;
    for(var j = 0; j < headers.length; j++){
      var h = String(headers[j] || '').trim();
      if(!h) continue;
      var v = values[i][j];
      if(v !== '' && v !== null) empty = false;
      o[h] = v;
    }
    if(empty) continue;
    (CONFIG.JSON_COLS[key] || []).forEach(function(c){
      if(typeof o[c] === 'string'){ try{ o[c] = JSON.parse(o[c]); }catch(e){} }
    });
    rows.push(o);
  }
  return rows;
}

function writeCollection_(key, rows){
  rows = rows || [];
  var headers = [];
  rows.forEach(function(r){
    Object.keys(r).forEach(function(k){ if(headers.indexOf(k) === -1) headers.push(k); });
  });
  if(!headers.length) headers = [CONFIG.ID_KEY[key] || 'id'];

  var data = rows.map(function(r){
    return headers.map(function(h){
      var v = (h in r) ? r[h] : '';
      if((CONFIG.JSON_COLS[key] || []).indexOf(h) > -1 && v && typeof v === 'object'){
        v = JSON.stringify(v);
      }
      return v;
    });
  });

  var sh = sheetFor_(key);
  sh.clear();
  sh.getRange(1, 1, data.length + 1, headers.length)
    .setValues([headers].concat(data));
  sh.getRange(1, 1, 1, headers.length).setFontWeight('bold');
  sh.setFrozenRows(1);
  return rows.length;
}
/* ═══════════════════════ OPERATIONS ═══════════════════════ */

function allCollections_(){
  var out = {};
  Object.keys(CONFIG.SHEETS).forEach(function(k){ out[k] = readCollection_(k); });
  return out;
}

function getRecord_(key, id){
  var idKey = CONFIG.ID_KEY[key] || 'id';
  var rows = readCollection_(key);
  for(var i = 0; i < rows.length; i++){
    if(String(rows[i][idKey]) === String(id)) return rows[i];
  }
  return null;
}

function appendRecord_(key, row){
  var rows = readCollection_(key);
  var idKey = CONFIG.ID_KEY[key] || 'id';
  if(!row[idKey]) row[idKey] = key.toUpperCase().slice(0,3) + '-' + Date.now();
  rows.push(row);
  writeCollection_(key, rows);
  return row;
}

function updateRecord_(key, id, patch){
  var rows = readCollection_(key);
  var idKey = CONFIG.ID_KEY[key] || 'id';
  var hit = null;
  for(var i = 0; i < rows.length; i++){
    if(String(rows[i][idKey]) === String(id)){
      Object.keys(patch || {}).forEach(function(k){ rows[i][k] = patch[k]; });
      hit = rows[i];
      break;
    }
  }
  if(hit) writeCollection_(key, rows);
  return hit;
}

function deleteRecord_(key, id){
  var rows = readCollection_(key);
  var idKey = CONFIG.ID_KEY[key] || 'id';
  var kept = rows.filter(function(r){ return String(r[idKey]) !== String(id); });
  writeCollection_(key, kept);
  return rows.length - kept.length;
}

function saveCollection_(key, rows){
  if(!(key in CONFIG.SHEETS)) throw new Error('Unknown collection: ' + key);
  return writeCollection_(key, rows);
}

function login_(email, password){
  var users = readCollection_('users');
  for(var i = 0; i < users.length; i++){
    var u = users[i];
    if(String(u.email).toLowerCase() === String(email || '').toLowerCase() &&
       String(u.password) === String(password || '')){
      return { id: u.id, name: u.name, email: u.email, role: u.role };
    }
  }
  return null;
}

/* Run ONCE from the editor (Run ▸ setup) to create sheets + seed. */
function setup(force){
  var created = [];
  Object.keys(CONFIG.SHEETS).forEach(function(k){
    if(force || readCollection_(k).length === 0){
      writeCollection_(k, SEED[k] || []);
      created.push(CONFIG.SHEETS[k]);
    }
  });
  Logger.log('Meridian DB ready → ' + ss().getUrl());
  Logger.log('Seeded: ' + (created.join(', ') || 'nothing (already had data)'));
  return { url: ss().getUrl(), seeded: created };
}
/* ═══════════════════════ WEB APP ENDPOINTS ═══════════════════════ */

function json_(obj){
  return ContentService.createTextOutput(JSON.stringify(obj))
                       .setMimeType(ContentService.MimeType.JSON);
}

function withLock_(fn){
  var lock = LockService.getScriptLock();
  lock.waitLock(10000);
  try{ return fn(); } finally { lock.releaseLock(); }
}

function handle_(req){
  try{
    switch(req.action){

      case 'health':
        return { ok:true, service:'Meridian Sheets API', time:new Date().toISOString() };

      case 'all':
        return { ok:true, data: allCollections_() };

      case 'get':
        return { ok:true, record: getRecord_(req.collection, req.id) };

      case 'save':                                   /* replace whole collection */
        return withLock_(function(){
          return { ok:true, count: saveCollection_(req.collection, req.rows || []) };
        });

      case 'append':
        return withLock_(function(){
          return { ok:true, record: appendRecord_(req.collection, req.row || {}) };
        });

      case 'update':
        return withLock_(function(){
          var r = updateRecord_(req.collection, req.id, req.patch || {});
          if(!r) throw new Error('Record not found: ' + req.id);
          return { ok:true, record: r };
        });

      case 'delete':
        return withLock_(function(){
          return { ok:true, removed: deleteRecord_(req.collection, req.id) };
        });

      case 'login': {
        var user = login_(req.email, req.password);
        if(!user) return { ok:false, error:'Invalid credentials' };
        return { ok:true, user:user };
      }

      case 'setup':
        return withLock_(function(){ return setup(!!req.force); });

      default:
        return { ok:false, error:'Unknown action: ' + req.action };
    }
  }catch(err){
    return { ok:false, error:String(err && err.message || err) };
  }
}

function doGet(e){
  var p = (e && e.parameter) ? e.parameter : {};
  if(!p.action) p.action = 'health';
  return json_(handle_(p));
}

function doPost(e){
  var body = {};
  try{ body = JSON.parse(e.postData.contents || '{}'); }catch(err){ body = {}; }
  if(!body.action && e && e.parameter && e.parameter.action){
    body = e.parameter;                          /* allow query-string fallback */
  }
  return json_(handle_(body));
}
